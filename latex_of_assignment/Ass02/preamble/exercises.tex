% --- Answer box, printed only in the answer key ---
% \begin{answer} ... \end{answer}
\newenvironment{answer}
  {\ifshowanswers
     \begin{tcolorbox}[
       breakable, enhanced,
       colback = black!3,
       colframe= black!45,
       boxrule = 0.4pt,
       left=6pt, right=6pt, top=4pt, bottom=4pt,
       title   = \bfseries\txtanswer,
       fonttitle = \normalsize,
     ]
   \else
     \setbox0\vbox\bgroup
   \fi}
  {\ifshowanswers
     \end{tcolorbox}
   \else
     \egroup
   \fi}

   \fi}
  {\ifshowanswers\else\end{minipage}\fi}

% --- Square grid -n..n with axes for plotting ---
% \coordgrid{}               empty -6..6
% \coordgrid[8]{\draw ...;}  -8..8 with a drawing on top
\newcommand{\coordgrid}[2][6]{%
  \begin{center}
  \begin{tikzpicture}[scale=0.48, every node/.style={font=\scriptsize}]
    \draw[black!18, thin] (-#1,-#1) grid (#1,#1);
    \draw[-{Stealth}, semithick] (-#1-0.4,0) -- (#1+0.7,0) node[right]{$X$};
    \draw[-{Stealth}, semithick] (0,-#1-0.4) -- (0,#1+0.7) node[above]{$Y$};
    \foreach \k in {-#1,...,-1,1,2,...,#1}{
      \node[below, black!60] at (\k,0) {$\k$};
      \node[left,  black!60] at (0,\k) {$\k$};
    }
    #2
  \end{tikzpicture}
  \end{center}}

% ===============================================================
% Points
%   \points{2}        prints "(2 คะแนน)" at the right edge and counts it
%   \totalpoints      sum of every \points in the document
%   \newpointgroup[title]  starts a new group (a sheet, an exam part)
%   \grouppoints      sum of the current group
% Totals come from the .aux file, so they show ?? until the second run;
% latexmk does that run by itself. Half points work: \points{1.5}
% ===============================================================
\makeatletter
\def\ex@total{0}
\def\ex@group{0}
\def\ex@grouptitle{}
\def\ex@ngroups{0}
\newcounter{ptgroup}

\newcommand{\points}[1]{%
  \xdef\ex@total{\fpeval{\ex@total+(#1)}}%
  \xdef\ex@group{\fpeval{\ex@group+(#1)}}%
  \unskip\nobreak\hfil\penalty50\hskip1em\hbox{}\nobreak\hfill
  \mbox{\small(#1 \txtpoints)}}

\newcommand{\ex@closegroup}{%
  \ifnum\value{ptgroup}>0
    \immediate\write\@auxout{\string\ex@savegroup{\the\value{ptgroup}}{\ex@group}%
      {\unexpanded\expandafter{\ex@grouptitle}}}%
  \fi}
\newcommand{\newpointgroup}[1][]{%
  \ex@closegroup
  \stepcounter{ptgroup}%
  \gdef\ex@group{0}%
  \gdef\ex@grouptitle{#1}}
% Called from the .aux file: \ex@savegroup{number}{points}{title}
\newcommand{\ex@savegroup}[3]{%
  \global\@namedef{ex@gpts@#1}{#2}%
  \global\@namedef{ex@gtitle@#1}{#3}%
  \ifnum#1>\ex@ngroups\relax\xdef\ex@ngroups{#1}\fi}
\AtEndDocument{%
  \ex@closegroup
  \immediate\write\@auxout{\gdef\string\ex@savedtotal{\ex@total}}}

\newcommand{\totalpoints}{\ifdefined\ex@savedtotal\ex@savedtotal\else??\fi}
\newcommand{\grouppoints}{%
  \@ifundefined{ex@gpts@\the\value{ptgroup}}{??}{\@nameuse{ex@gpts@\the\value{ptgroup}}}}
\makeatother
